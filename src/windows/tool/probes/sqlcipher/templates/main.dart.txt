import 'dart:convert';
import 'dart:io';
import 'package:flutter/material.dart';
import 'package:miyu_sqlcipher_probe/probe.dart';

void main() => runApp(const MaterialApp(home: SqlcipherProbeApp()));

class SqlcipherProbeApp extends StatefulWidget {
  const SqlcipherProbeApp({super.key});
  @override
  State<SqlcipherProbeApp> createState() => _SqlcipherProbeAppState();
}

class _SqlcipherProbeAppState extends State<SqlcipherProbeApp> {
  String result = '准备验证，仅使用人工数据';
  @override
  void initState() {
    super.initState();
    Future<void>.delayed(Duration.zero, () {
      try {
        final dir = Directory.systemTemp.createTempSync('miyu-sqlcipher-');
        final report = runCoreProbe(dir);
        if (mounted) setState(() => result = 'PASS\n${const JsonEncoder.withIndent('  ').convert(report)}');
      } catch (e) {
        if (mounted) setState(() => result = 'FAIL: ${e is ProbeFailure ? e.check : e.runtimeType}');
      }
    });
  }
  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: const Text('SQLCipher 隔离验证')),
    body: SingleChildScrollView(padding: const EdgeInsets.all(16), child: SelectableText(result)),
  );
}
