import 'dart:convert';
import 'dart:io';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:miyu_sqlcipher_probe/probe.dart';

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();
  testWidgets('SQLCipher 原生加密、日志与同进程恢复', (tester) async {
    final directory = Directory.systemTemp.createTempSync('miyu-sqlcipher-test-');
    final report = runCoreProbe(directory);
    expect(report['checks'], hasLength(11));
    expect(report['cipher'].toString(), startsWith('4.20.0'));
    binding.reportData = report;
    print(jsonEncode(report));
  });
}
